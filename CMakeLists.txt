cmake_minimum_required(VERSION 3.24)
project(BLPFormat2 VERSION 2.0.3 LANGUAGES C CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "BLPFormat2 requires 64-bit Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS)
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/utf-8>)
add_compile_options($<$<COMPILE_LANGUAGE:RC>:/c65001>)

file(GLOB JPEG_SOURCES CONFIGURE_DEPENDS third_party/jpeg/*.c)
add_library(jpeg STATIC ${JPEG_SOURCES})
target_include_directories(jpeg PUBLIC third_party/jpeg)
add_library(blp_codec STATIC src/BlpCodec.cpp third_party/reference/BlpDecoder.cpp)
target_include_directories(blp_codec PUBLIC src third_party/reference)
target_link_libraries(blp_codec PUBLIC jpeg windowscodecs ole32)

set(SDK "${CMAKE_CURRENT_SOURCE_DIR}/third_party/sdk/photoshopapi")
set(SDK_INCLUDES "${SDK}/photoshop" "${SDK}/pica_sp" "${SDK}/resources")
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/BLPFormat2.pipl"
    COMMAND ${CMAKE_COMMAND} -DCL=${CMAKE_C_COMPILER} -DSDK=${SDK}
        -DSOURCE=${CMAKE_CURRENT_SOURCE_DIR}/resources/Plugin.r
        -DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/BLPFormat2.pipl
        -DCONVERTER=${CMAKE_CURRENT_SOURCE_DIR}/tools/Cnvtpipl.exe
        -P ${CMAKE_CURRENT_SOURCE_DIR}/tools/Pipl.cmake
    DEPENDS resources/Plugin.r tools/Pipl.cmake VERBATIM)
add_custom_target(pipl DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/BLPFormat2.pipl")
add_library(BLPFormat2 MODULE src/Plugin.cpp src/ExportDialog.cpp resources/Plugin.rc)
add_dependencies(BLPFormat2 pipl)
target_include_directories(BLPFormat2 PRIVATE ${SDK_INCLUDES} ${CMAKE_CURRENT_BINARY_DIR} resources)
target_compile_definitions(BLPFormat2 PRIVATE MSWindows=1)
target_link_libraries(BLPFormat2 PRIVATE blp_codec comctl32 uxtheme dwmapi)
set_target_properties(BLPFormat2 PROPERTIES PREFIX "" SUFFIX ".8bi")

add_executable(blp_tests tests/CodecTests.cpp)
target_link_libraries(blp_tests PRIVATE blp_codec)
add_executable(BLPFormat2Settings WIN32 tools/SettingsMain.cpp src/ExportDialog.cpp resources/Dialog.rc)
target_include_directories(BLPFormat2Settings PRIVATE src resources third_party/reference)
target_link_libraries(BLPFormat2Settings PRIVATE blp_codec comctl32 uxtheme dwmapi)
include(CTest)
add_test(NAME codec COMMAND blp_tests)
add_executable(host_tests tests/HostTests.cpp)
target_include_directories(host_tests PRIVATE ${SDK_INCLUDES})
target_compile_definitions(host_tests PRIVATE MSWindows=1)
target_link_libraries(host_tests PRIVATE blp_codec)
add_dependencies(host_tests BLPFormat2)
add_test(NAME photoshop_host_contract COMMAND host_tests $<TARGET_FILE:BLPFormat2>)
add_executable(read_tests tests/ReadTests.cpp)
target_include_directories(read_tests PRIVATE ${SDK_INCLUDES})
target_compile_definitions(read_tests PRIVATE MSWindows=1)
target_link_libraries(read_tests PRIVATE blp_codec)
add_dependencies(read_tests BLPFormat2)
add_test(NAME read_transfer COMMAND read_tests $<TARGET_FILE:BLPFormat2>)
add_executable(settings_tests tests/SettingsTests.cpp src/ExportDialog.cpp)
target_include_directories(settings_tests PRIVATE resources)
target_link_libraries(settings_tests PRIVATE blp_codec comctl32 uxtheme dwmapi)
add_dependencies(settings_tests BLPFormat2)
add_test(NAME persistent_settings COMMAND settings_tests $<TARGET_FILE:BLPFormat2>)
set_tests_properties(photoshop_host_contract persistent_settings PROPERTIES TIMEOUT 20)
add_executable(write_tests tests/WriteTests.cpp)
target_include_directories(write_tests PRIVATE ${SDK_INCLUDES})
target_compile_definitions(write_tests PRIVATE MSWindows=1)
target_link_libraries(write_tests PRIVATE blp_codec)
add_dependencies(write_tests BLPFormat2)
add_test(NAME write_transfer COMMAND write_tests $<TARGET_FILE:BLPFormat2>)
set_tests_properties(write_transfer PROPERTIES TIMEOUT 60)
